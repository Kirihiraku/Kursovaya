CREATE TABLE Users (
    UserId INT PRIMARY KEY AUTO_INCREMENT,
    Username VARCHAR(100) NOT NULL
);

CREATE TABLE QRCodes (
    QRCodeId INT PRIMARY KEY AUTO_INCREMENT,
    UserId INT,
    QRCodeText TEXT,
    CreatedAt DATETIME,
    FOREIGN KEY (UserId) REFERENCES Users(UserId)
);
